```python
def _metric_score(self, prompt) -> str:
    answer   = _extract_section(prompt, "[답변]")
    contexts = _extract_section(prompt, "[컨텍스트]") or _extract_section(prompt, "[검색된 컨텍스트]")
    query    = _extract_section(prompt, "[질문]")
    ref = contexts or query
    a, c = _tokens(answer), _tokens(ref)
    overlap = len(a & c) / len(a) if a else 0.8
    score = round(min(1.0, 0.5 + overlap/2), 2)  # 0.5~1.0
    return json.dumps({"score": score, "reasoning": "mock 휴리스틱 평가", "grounding_chunks": []})
```

#### `deterministic_embedding(text, dim=64)`

```python
_TOKEN_RE = re.compile(r"[0-9A-Za-z가-힣]{2,}")

def _tokens(text) -> set[str]:
    base = set(_TOKEN_RE.findall(text.lower()))
    compact = re.sub(r"\s+", "", text)
    base |= {compact[i:i+2] for i in range(len(compact)-1)}  # 2-gram
    return base

def deterministic_embedding(text, dim=64) -> list[float]:
    vec = [0.0] * dim
    for tok in _tokens(text) or {text}:
        h = int(hashlib.md5(tok.encode("utf-8")).hexdigest(), 16)
        vec[h % dim] += 1.0
    norm = math.sqrt(sum(v*v for v in vec)) or 1.0
    return [v/norm for v in vec]  # L2 정규화
```

### 2.7 `core/store.py` — SQLite ResultStore

#### 핵심 설계 원칙

1. **scores 분리 테이블**: 메트릭별 별도 행 저장 → `WHERE metric='faithfulness' AND score < 0.7` SQL 필터
2. **WAL 모드**: `PRAGMA journal_mode=WAL` → 동시 읽기/쓰기 허용

class Tracer:
    def __init__(self, run_id: str, results_dir: str = "data/results"):
        self.base_dir = Path(results_dir) / run_id

    def save(self, name: str, data: Any) -> Path:

class EvaluationRouter:
    def needs_escalation(self, record) -> bool:
        """저신뢰 → 추가 샘플/상위 모델 필요 판단"""
        if not opts.escalate_on_low_confidence: return False
        return any(s.is_low_confidence for s in record.scores)

    def select_for_human_review(self, records) -> list[EvaluationRecord]:
        """Active sampling: 우선순위 점수 기반 상위 quota 선정"""
        scored = [(self._priority(r), r) for r in records]
        scored.sort(key=lambda x: x[0], reverse=True)
        selected = [r for _, r in scored[:opts.active_sampling_quota]]
        for r in selected:
            r.needs_human_review = True  # ← side effect
        return selected

    def _priority(self, record) -> float:
        """불확실성 + SLA 경계 근접도"""
        uncertainty = 0.0
        boundary = 0.0
        for s in record.scores:
            uncertainty += (1.0 - s.confidence)
            if s.is_low_confidence: uncertainty += 0.5
            thr = self.sla.get(s.metric)
            if thr is not None:
                dist = abs(s.score - thr)
                if dist < 0.1:
                    boundary += (0.1 - dist) * 10  # 0~1 범위
        n = max(1, len(record.scores))
        return uncertainty/n + boundary/n
        # SLA 경계(±0.1) 근처 항목이 높은 우선순위
```

        # → "고객: ...\n콜봇: ...\n시스템: ..."  (role → 한국어 매핑)

            MetricScore(metric="efficiency",    score=efficiency_score, method="session"),
            MetricScore(metric="escalation_handling",
                        score=0.0 if escalation_needed else 1.0, method="session"),
        ]

        # multiturn_consistency → DomainMetricsEvaluator 재사용
        dm = DomainMetricsEvaluator(provider, DomainMetricsOptions())
        scores.append(await dm.multiturn_consistency(call_log))

        if ground_truth:
            # 마지막 봇 답변 vs GT 임베딩 cosine 유사도
            scores.append(await self._gt_similarity(call_log, ground_truth))

        return SessionEvaluation(
            conversation_id=call_log.call_id,
            tenant_id=call_log.metadata.get("tenant_id", "default"),
            scores=scores, resolved=resolved, escalation_needed=escalation_needed,
            judge_provider=type(provider).__name__.replace("Provider","").lower(),
        )

    @staticmethod
    def _cosine(a, b) -> float:
        dot = sum(x*y for x,y in zip(a,b))
        na = math.sqrt(sum(x*x for x in a)) or 1.0
        nb = math.sqrt(sum(y*y for y in b)) or 1.0
        return dot / (na * nb)
```

class HeuristicClassifier:
    """토큰 중첩 기반 메트릭 근사 (LLM 호출 없음)"""

    def faithfulness(self, pair) -> float:
        ans = _tokens(pair.bot_answer)
        ctx = {t for c in retrieval.contexts for t in _tokens(c.content)}
        return round(len(ans & ctx) / len(ans), 4) if ans else 0.0

    def context_recall(self, pair) -> float:
        q = _tokens(pair.question)
        ctx = {t for c in retrieval.contexts for t in _tokens(c.content)}
        return round(len(q & ctx) / len(q), 4) if q else 0.0

    def score(self, pair, metric) -> float:
        if metric == "faithfulness": return self.faithfulness(pair)
        if metric == "context_recall": return self.context_recall(pair)
        # 그 외: bot_answer ↔ question 중첩 (answer_relevance 근사)
        return round(len(a & q) / len(a), 4) if a else 0.0

class MetaEvaluator:
    def load_golden(self) -> dict[(qa_id, metric), float]:
        """JSONL: {"qa_id":"...","metric":"faithfulness","human_score":0.75}"""

    def _spearman(self, a, b) -> float:
        """동점 처리: 평균 순위 (tied rank)"""
        ra, rb = self._rank(a), self._rank(b)
        cov = Σ(ra_i - mean_ra)(rb_i - mean_rb)
        return cov / (std_ra * std_rb)  # Pearson of ranks

    def _cohen_kappa(self, a, b) -> float:
        """5-bin 이산화: _bin(score) = min(4, int(round(score*4)))"""
        # 0.0→0, 0.25→1, 0.5→2, 0.75→3, 1.0→4
        po = 관찰 일치율
        pe = 기대 일치율 = Σ(pa_k * pb_k)
        return (po - pe) / (1 - pe)

    def _stats(self, pairs) -> dict:
        # opts.metrics에 포함된 항목만 계산
        if "mae" in opts.metrics: out["mae"] = mean(|judge-human| for ...)
        if "spearman" in opts.metrics: out["spearman"] = _spearman(judge, human)
        if "kappa" in opts.metrics: out["kappa"] = _cohen_kappa(judge, human)

    rng = random.Random(seed)  # 결정적 (seed 고정)

    hi_idx = min(n_samples-1, int((1-alpha/2) * n_samples))

def paired_permutation_test(current, previous, n_permutations=1000, alpha=0.05, seed=42):
    observed = mean(current) - mean(previous)
    combined = current + previous
    count = sum(1 for _ in range(n_permutations) if |perm_delta| >= |observed|)
    p_value = (count+1) / (n_permutations+1)  # 스무딩 (zero count 방지)
    return SignificanceResult(delta=observed, p_value=p_value, significant=(p_value<alpha))

def ppi_mean(classifier_scores, labeled_pairs, confidence=0.95, seed=42):
    # labeled_pairs = [(classifier_score, llm_judge_score), ...]
    clf_mean = mean(classifier_scores)
    rectifiers = [judge - clf for clf, judge in labeled_pairs]
    rect_ci = bootstrap_ci(rectifiers, confidence=confidence, seed=seed)

        point_estimate=point,

        ci_high=clf_mean + rect_ci.high,
        naive_classifier_mean=clf_mean,
