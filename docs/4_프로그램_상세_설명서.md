```python
def _is_transient(exc: BaseException) -> bool:
    # 재시도 대상: Rate Limit, Timeout, 서버 오류
    transient_markers = ("ratelimit", "timeout", "apiconnection", "internalserver", "serviceunavailable")
    if any(m in type(exc).__name__.lower() for m in transient_markers):
        return True
    status = getattr(exc, "status_code", None) or getattr(exc, "status", None)
    return status in (408, 409, 429, 500, 502, 503, 504)
    # 400(Bad Request)은 재시도 없음 → 코드 버그 즉시 노출
```

```python
async def gather_with_concurrency(coros, concurrency=5) -> list:
    """asyncio.Semaphore로 동시 실행 수 제한"""
    semaphore = asyncio.Semaphore(concurrency)
    async def bounded(coro):

    return list(await asyncio.gather(*[bounded(c) for c in coros]))
```

```python
def _metric_score(self, prompt) -> str:
    """토큰 중첩 기반 0.5~1.0 범위 점수 생성"""
    answer = _extract_section(prompt, "[답변]")
    contexts = _extract_section(prompt, "[컨텍스트]") or _extract_section(prompt, "[검색된 컨텍스트]")
```

#### `deterministic_embedding(text, dim=64)` — 결정적 임베딩

```python
def deterministic_embedding(text: str, dim: int = 64) -> list[float]:
    """텍스트 토큰을 MD5 해시 버킷에 매핑 → L2 정규화 (64차원)"""
    vec = [0.0] * dim
    for tok in _tokens(text) or {text}:
        h = int(hashlib.md5(tok.encode("utf-8")).hexdigest(), 16)
        vec[h % dim] += 1.0
    norm = math.sqrt(sum(v*v for v in vec)) or 1.0
    return [v / norm for v in vec]

# _tokens(): 2자 이상 알파벳/한글 토큰 + 2-gram 문자 bigram (공백 제거 후)
_TOKEN_RE = re.compile(r"[0-9A-Za-z가-힣]{2,}")
def _tokens(text: str) -> set[str]:
    base = set(_TOKEN_RE.findall(text.lower()))
    compact = re.sub(r"\s+", "", text)
    base |= {compact[i:i+2] for i in range(len(compact)-1)}
    return base
```

> **결정적 보장**: 동일 텍스트 → 동일 MD5 → 동일 64차원 벡터. CI 재현성 보장.

### 2.8 `cp2_knowledge_base/chunker.py` — Parent-Child 청커

```python
class ParentChildChunker:
    parent_size = 1000  # words
    child_size  = 200   # words
    overlap     = 50    # words

    def _build_full_text(self, call_log: CallLog) -> str:
        # 각 턴: f"[{turn.role.upper()}] {turn.content}"
        # → "[USER] 질문\n[BOT] 답변\n..."
        return "\n".join(parts)

    @staticmethod
    def _split_text(text: str, chunk_size: int, overlap: int) -> list[str]:
        words = text.split()
        chunks = []
        start = 0
        while start < len(words):
            end = min(start + chunk_size, len(words))
            chunks.append(" ".join(words[start:end]))
            if end >= len(words): break
            start += chunk_size - overlap  # 슬라이딩 (중복 포함)
        return chunks

    @staticmethod
    def _approx_tokens(text: str) -> int:
        return len(text.split())  # 단어 수 = 토큰 수 근사
```

```python
def _aggregate_samples(self, metric: str, samples: list[MetricScore]) -> MetricScore:
    valid = [s for s in samples if not s.reasoning.startswith("Parse error")]
    if not valid: valid = samples  # 전부 실패해도 기록

    sample_scores = [s.score for s in valid]
    median_score = statistics.median(sample_scores)
    std = statistics.pstdev(sample_scores) if len(sample_scores) > 1 else 0.0
    confidence = max(0.0, 1.0 - std / 0.5)  # std=0 → 1.0, std=0.5 → 0.0, std>0.5 → 0.0
    is_low = std > self.low_confidence_std  # default: 0.2

    # 중앙값에 가장 가까운 샘플의 reasoning 채택 (대표 근거)
    best = min(valid, key=lambda s: abs(s.score - median_score))

    return MetricScore(
        score=round(median_score, 4),
        confidence=round(confidence, 4),
        sample_scores=sample_scores,
        is_low_confidence=is_low,
        reasoning=best.reasoning,
        grounding_chunks=best.grounding_chunks,
    )
```

### 2.17 `cp5_aggregator/statistics_ext.py` — 통계 확장

```python
# 1. 부트스트랩 신뢰구간 (stdlib random만 사용)
@dataclass
class ConfidenceInterval:
    mean: float; low: float; high: float; n: int

def bootstrap_ci(values, n_samples=1000, confidence=0.95, seed=42) -> ConfidenceInterval:
    rng = random.Random(seed)  # 결정적 (seed 고정)
    means = [sum(rng.choices(values, k=len(values)))/len(values) for _ in range(n_samples)]
    means.sort()
    alpha = 1 - confidence
    lo_idx = int((alpha/2) * n_samples)
    hi_idx = min(n_samples-1, int((1-alpha/2) * n_samples))
    return ConfidenceInterval(mean=mean(values), low=means[lo_idx], high=means[hi_idx], n=n)

# 2. 순열검정 (paired permutation test, 비모수)
@dataclass
class SignificanceResult:
    delta: float; p_value: float; significant: bool; ci_low: float; ci_high: float

def paired_permutation_test(current, previous, n_permutations=1000, alpha=0.05, seed=42):
    observed = mean(current) - mean(previous)
    combined = current + previous
    count = sum(1 for _ in n_permutations if |perm_delta| >= |observed|)
    p_value = (count+1) / (n_permutations+1)  # 스무딩
    return SignificanceResult(delta=observed, p_value=p_value, significant=(p_value<alpha), ...)

# 3. PPI 추정 (Prediction-Powered Inference)
@dataclass
class PPIEstimate:
    point_estimate: float; ci_low: float; ci_high: float
    n_labeled: int; n_total: int; naive_classifier_mean: float

def ppi_mean(classifier_scores, labeled_pairs, confidence=0.95, seed=42) -> PPIEstimate:
    # estimate = mean(classifier_all) + mean(judge_labeled - classifier_labeled)
    # rectifiers = [judge - clf for clf, judge in labeled_pairs]
    # point = clf_mean + bootstrap_ci(rectifiers).mean
    clf_mean = mean(classifier_scores)
    rectifiers = [judge-clf for clf,judge in labeled_pairs]
    rect_ci = bootstrap_ci(rectifiers, confidence=confidence, seed=seed)
    point = clf_mean + rect_ci.mean
    return PPIEstimate(point_estimate=point, ci_low=clf_mean+rect_ci.low, ...)
```

```python
    for p in pairs:
```
